from pathlib import Path
import json,hashlib,collections
r=Path("/tmp/research-agent-desktop-phase02-final-6e7ayl45")
repo=Path("/home/minervamuses/research-agent-workspace")
a=[json.loads(l) for l in (r/"native-audit.jsonl").read_text().splitlines()]
doc=json.loads((r/"store/conversations/28b222e0cc6543aa8d7bbdc423de99a7.json").read_text())
citation=doc["turns"][3]
result=next(x for x in a if x["kind"]=="result" and x["method"]=="session.turn" and x["result"]["turnId"]==citation["turnId"])
assert citation["assistantOutput"]==result["result"]["text"]
assert citation["thinkingMode"]=="normal" and citation["state"]=="completed"
assert [x["callId"] for x in citation["toolActivities"]]==["search-1","save-1"]
assert result["model_calls"]==3 and result["fetch_calls"]==4 and result["scopes"]==1
assert [t["state"] for t in doc["turns"][-3:]]==["failed","interrupted","completed"]
assert doc["turns"][-1]["assistantOutput"]=="ordinary answer"
before=json.loads((r/"bundle-before-lifecycle.json").read_text())
after={str(p.relative_to(r)):dict(sha256=hashlib.sha256(p.read_bytes()).hexdigest(),mtime_ns=p.stat().st_mtime_ns) for p in (r/"cite").rglob("*") if p.is_file()}
assert before==after
assert {x["result"]["text"] for x in a if x["kind"]=="result" and x["method"]=="session.turn" and "ordinary" in x["result"].get("text","")}=={"ordinary answer"}
restart_results=[x for x in a if x["pid"]==14577 and x["kind"]=="result" and x["method"] in ("session.select","session.transcript")]
assert restart_results and all(x["model_calls"]==x["fetch_calls"]==x["scopes"]==0 for x in restart_results)
selections=[x["result"] for x in a if x["kind"]=="result" and x["method"]=="session.select"]
assert selections[0]["loadedSkills"]==["citation"]
assert all("native-writer" in x["loadedSkills"] for x in selections[1:3])
assert all(x["loadedSkills"]==["citation"] and x["extensionRevision"]==2 for x in selections[3:])
requests=[x for x in a if x["kind"]=="request" and x["method"]=="session.turn"]
assert len(requests)==10
assert len([x for x in requests if x["params"]["text"]=="/status "])==1
print("PASS native canonical/final answer, normal mode, tool activities, ordinary next, lifecycle counters, catalog revisions, bundle hash+mtime, failed/interrupted recovery; 10 UI session.turn requests.")
print("Bundle hashes",json.dumps(after))
